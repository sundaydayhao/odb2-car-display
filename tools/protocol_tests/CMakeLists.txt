cmake_minimum_required(VERSION 3.16)
project(elm_protocol_tests C)
set(CMAKE_C_STANDARD 99)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(elm_protocol_tests test_protocol.c
    "${ROOT}/main/elm327_protocol.c" "${ROOT}/main/obd2_decode.c")
target_include_directories(elm_protocol_tests PRIVATE "${ROOT}/main")
if(NOT MSVC)
    target_compile_options(elm_protocol_tests PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_executable(media_tests test_media.c "${ROOT}/main/media_model.c")
target_include_directories(media_tests PRIVATE "${ROOT}/main")
if(NOT MSVC)
    target_compile_options(media_tests PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME media COMMAND media_tests)
add_executable(wifi_networks_tests test_wifi_networks.c "${ROOT}/main/wifi_networks.c")
target_include_directories(wifi_networks_tests PRIVATE "${ROOT}/main")
add_test(NAME wifi_networks COMMAND wifi_networks_tests)
add_test(NAME elm_protocol COMMAND elm_protocol_tests)
set(CJSON "${ROOT}/managed_components/espressif__cjson/cJSON")
add_executable(ota_policy_tests test_ota_policy.c "${ROOT}/main/ota_policy.c" "${CJSON}/cJSON.c")
target_include_directories(ota_policy_tests PRIVATE "${ROOT}/main" "${CJSON}")
add_test(NAME ota_policy COMMAND ota_policy_tests)
add_executable(replay_response replay_response.c "${ROOT}/main/elm327_protocol.c" "${ROOT}/main/obd2_decode.c")
target_include_directories(replay_response PRIVATE "${ROOT}/main")
