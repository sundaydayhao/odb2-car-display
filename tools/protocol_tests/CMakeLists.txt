cmake_minimum_required(VERSION 3.16)
project(elm_protocol_tests C)
set(CMAKE_C_STANDARD 99)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(DISPLAY_MAIN "${ROOT}/display_board/main")
add_executable(elm_protocol_tests test_protocol.c
    "${DISPLAY_MAIN}/elm327_protocol.c" "${DISPLAY_MAIN}/obd2_decode.c")
target_include_directories(elm_protocol_tests PRIVATE "${DISPLAY_MAIN}")
if(NOT MSVC)
    target_compile_options(elm_protocol_tests PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_executable(photo_bmp_tests test_photo_bmp.c "${DISPLAY_MAIN}/photo_bmp.c")
target_include_directories(photo_bmp_tests PRIVATE "${DISPLAY_MAIN}")

    target_compile_options(photo_bmp_tests PRIVATE -Wall -Wextra -Werror)
